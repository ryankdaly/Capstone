### AbsoluteValue — pure function, no preconditions needed
method AbsoluteValue(x: int) returns (result: int)
  ensures result >= 0
  ensures result == x || result == -x
{
  if x >= 0 { result := x; } else { result := -x; }
}

### Clamp — branching with range postconditions
method Clamp(value: int, lo: int, hi: int) returns (result: int)
  requires lo <= hi
  ensures lo <= result <= hi
  ensures value < lo        ==> result == lo
  ensures value > hi        ==> result == hi
  ensures lo <= value <= hi ==> result == value
{
  if value < lo {
    result := lo;
  } else if value > hi {
    result := hi;
  } else {
    result := value;
  }
}

### SafeAdd — overflow detection, multiple return values, const declarations
const INT32_MAX: int := 2147483647
const INT32_MIN: int := -2147483648

method SafeAdd(a: int, b: int) returns (result: int, ok: bool)
  requires INT32_MIN <= a <= INT32_MAX
  requires INT32_MIN <= b <= INT32_MAX
  ensures ok  ==> result == a + b && INT32_MIN <= result <= INT32_MAX
  ensures !ok ==> (a + b > INT32_MAX || a + b < INT32_MIN)
{
  if a + b > INT32_MAX || a + b < INT32_MIN {
    result := 0;
    ok := false;
  } else {
    result := a + b;
    ok := true;
  }
}

### Sigmoid — transcendental math: spec the PROPERTY, not the computation
// Dafny has no exp(). Spec what sigmoid guarantees: output in (0,1),
// monotonically increasing, symmetric around 0.5 at x=0.
method Sigmoid(x: real) returns (result: real)
  ensures 0.0 < result < 1.0
  ensures x == 0.0 ==> result == 0.5
  ensures x > 0.0  ==> result > 0.5
  ensures x < 0.0  ==> result < 0.5
{
  // Body approximates 1/(1+exp(-x)) via a rational approximation verifiable
  // against the postconditions. Dafny checks the ensures, not the formula.
  if x >= 8.0 {
    result := 0.9999;
  } else if x <= -8.0 {
    result := 0.0001;
  } else {
    // Linear approximation in [-8,8] that satisfies all postconditions
    result := 0.5 + x / 16.0 + x * x * x / 12288.0;
    if result >= 1.0 { result := 0.9999; }
    if result <= 0.0 { result := 0.0001; }
  }
}

### Factorial — recursive with decreases clause
method Factorial(n: int) returns (result: int)
  requires n >= 0
  ensures result >= 1
  decreases n
{
  if n == 0 {
    result := 1;
  } else {
    var sub := Factorial(n - 1);
    result := n * sub;
  }
}

### MinValue — array minimum, full postcondition
method MinValue(a: array<int>) returns (min: int)
  requires a != null && a.Length > 0
  ensures forall k :: 0 <= k < a.Length ==> min <= a[k]
  ensures exists k :: 0 <= k < a.Length && min == a[k]
{
  min := a[0];
  var i := 1;
  while i < a.Length
    invariant 1 <= i <= a.Length
    invariant forall k :: 0 <= k < i ==> min <= a[k]
    invariant exists k :: 0 <= k < i && min == a[k]
    decreases a.Length - i
  {
    if a[i] < min { min := a[i]; }
    i := i + 1;
  }
}

### RealArithmetic — int-to-real casting, division, bounds
method NormalizeToUnit(x: int, lo: int, hi: int) returns (result: real)
  requires lo < hi
  requires lo <= x <= hi
  ensures 0.0 <= result <= 1.0
{
  result := (x - lo) as real / (hi - lo) as real;
}

### IntMovingAverage.get() — class decomposed to standalone method (R15 pattern)
// Actor writes: class IntMovingAverage with __init__(window_size) and get() -> int.
// R15: model class state as parameters. Constructor arg → requires. Buffer → seq<int>.
method ComputeAverage(samples: seq<int>, window_size: int) returns (result: int)
  requires 1 <= window_size <= 256
  requires |samples| == window_size
  requires forall k :: 0 <= k < |samples| ==> -32768 <= samples[k] <= 32767
  ensures -32768 <= result <= 32767
{
  var sum := 0;
  var i := 0;
  while i < |samples|
    invariant 0 <= i <= |samples|
    invariant -32768 * i <= sum <= 32767 * i
    decreases |samples| - i
  {
    sum := sum + samples[i];
    i := i + 1;
  }
  result := sum / window_size;
}

R19. Dafny has NO ternary operator `? :`. Use expression-form if-then-else instead.
     WRONG:  (k == 0 ? 0 : prev[k])   → "closeparen expected"
     RIGHT:  (if k == 0 then 0 else prev[k])
     This applies in ensures, requires, and loop invariants — anywhere an expression appears.

R20. Ensures/requires clauses must be PURE BOOLEAN EXPRESSIONS — no statements allowed.
     Never put `var`, assignments, or `while` loops inside an ensures/requires.
     WRONG:  ensures ok ==> ( var x := 0; while ... { x := x ^ a[i]; } x == 0 )
     RIGHT:  If you need a loop-computed value in ensures, either:
             (a) spec a property (bounds/range) like the Sigmoid example, OR
             (b) define a pure `function` outside the method and reference it.
     Attempting to re-compute a loop inside ensures → "invalid UnaryExpression".

R16. Always parenthesize the right-hand side of ==> when it spans multiple lines or
     contains &&/||. Dafny warns "unusual indentation in right-hand operand of ==>" and
     may misparse without the parens.
     WRONG:
       ensures ok ==>
         condition1 &&
         condition2
     RIGHT:
       ensures ok ==> (
         condition1 &&
         condition2
       )

R17. Ensures predicates must mirror the body's exact computation. Never substitute a
     simpler predicate that is only equivalent under special assumptions.
     If the body uses `a - b <= tol && b - a <= tol`, the ensures MUST use that exact
     expression — NOT `a == b` (only equivalent when tol == 0). The verifier checks the
     ensures against every execution path; an overly strong postcondition causes
     "postcondition could not be proved".

R18. When you get "postcondition could not be proved on this return path": insert an
     `assert <predicate>;` statement immediately before the return/assignment on that
     branch. This gives the SMT solver an explicit hint on the path it cannot close.
     Example: if `ensures result >= 0` fails on the else-branch, add:
       assert result >= 0;   // SMT hint
       result := computed_value;

### TMRVote — majority voter, parenthesized ==> and mirrored tolerance predicates
// R16: wrap multi-line ==> RHS in parens.  R17: ensures mirrors body's tolerance check.
method TMRVote(a: int, b: int, c: int, tolerance: int) returns (result: int, ok: bool)
  requires tolerance >= 0
  ensures ok ==> (
    ((a - b <= tolerance && b - a <= tolerance) && result == a) ||
    ((a - c <= tolerance && c - a <= tolerance) && result == a) ||
    ((b - c <= tolerance && c - b <= tolerance) && result == b)
  )
  ensures !ok ==> !(
    (a - b <= tolerance && b - a <= tolerance) ||
    (a - c <= tolerance && c - a <= tolerance) ||
    (b - c <= tolerance && c - b <= tolerance)
  )
{
  if a - b <= tolerance && b - a <= tolerance {
    result := a; ok := true;
  } else if a - c <= tolerance && c - a <= tolerance {
    result := a; ok := true;
  } else if b - c <= tolerance && c - b <= tolerance {
    result := b; ok := true;
  } else {
    result := 0; ok := false;
  }
}
